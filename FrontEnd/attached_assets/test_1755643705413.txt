    async testEnrollmentProgress() {

        try {
            const enrollment = this.testEnrollments[0];
            
            // Simulate progress updates
            const progressUpdates = [25, 50, 75, 100];
            
            for (const progress of progressUpdates) {
                const isCompleted = progress === 100;
                const updateData = {
                    progress_percentage: progress,
                    is_completed: isCompleted
                };
                
                if (isCompleted) {
                    updateData.completion_date = new Date().toISOString().split('T')[0];
                }
                
                await this.pb.collection('course_enrollments').update(enrollment.id, updateData);
                console.log(`  ✓ Progress updated to ${progress}%`);

            // Find completed enrollments
            const completedEnrollments = await this.pb.collection('course_enrollments').getFullList({
                filter: 'is_completed = true',
                expand: 'course_id,user_id'
            });
            console.log(`  ✓ Completed enrollments query successful (${completedEnrollments.length} completed)`);
            
            // Verify final state
            const finalEnrollment = await this.pb.collection('course_enrollments').getOne(enrollment.id);
            assert.strictEqual(finalEnrollment.is_completed, true, 'Enrollment should be completed');

            // Simulate multiple quiz attempts
            for (let attempt = 1; attempt <= 3; attempt++) {
                const score = Math.floor(Math.random() * 40) + 60; // Random score 60-100

                await this.pb.collection('quizzes').update(quiz.id, {
                    tried_attemps: attempt,
                    is_completed: passed
                });
                
                console.log(`  ✓ Attempt ${attempt}: Score ${score}% (${passed ? 'PASSED' : 'FAILED'})`);

    async generateLearningSummary(userId) {
        try {
            const [enrollments, certificates, badges, activities] = await Promise.all([
                this.pb.collection('course_enrollments').getFullList({
                    filter: `user_id = "${userId}"`,
                    expand: 'course_id'
                }),
                this.pb.collection('certificates').getFullList({
                    filter: `user_id = "${userId}"`
                }),
                this.pb.collection('user_badges').getFullList({
                    filter: `user_id = "${userId}"`
                }),
                this.pb.collection('user_activity_logs').getList(1, 50, {
                    sort: '-created'
                })
            ]);
            
            const completedCourses = enrollments.filter(e => e.is_completed).length;
            const totalProgress = enrollments.reduce((sum, e) => sum + (e.progress_percentage || 0), 0);
            const avgProgress = enrollments.length > 0 ? totalProgress / enrollments.length : 0;
            
            return {
                total_enrollments: enrollments.length,
                completed_courses: completedCourses,
                certificates_earned: certificates.length,
                badges_earned: badges.length,
                average_progress: Math.round(avgProgress),
                recent_activities: activities.items.length
            };
        } catch (error) {
            console.log('Error generating learning summary:', error.message);
            return null;
        }
    }

        ];
        
        for (const collection of collections) {
            try {
                // Clean specific test items
                for (const item of collection.items) {
                    try {
                        await this.pb.collection(collection.name).delete(item.id);
                    } catch (e) {
                        // Item might already be deleted due to cascade
                    }
                }
                
                // Clean any remaining test data created today
                const today = new Date().toISOString().split('T')[0];
                const testItems = await this.pb.collection(collection.name).getFullList({
                    filter: `created >= "${today}"`
                }).catch(() => []);
                
                for (const item of testItems) {
                    try {
                        // Only delete items that look like test data
                        if (item.email?.includes('test_') || 
                            item.first_name?.includes('Test') || 
                            item.title?.includes('Test') ||
                            item.email?.includes('bulk_user_')) {
                            await this.pb.collection(collection.name).delete(item.id);
                        }
                    } catch (e) {
                        // Item might already be deleted
                    }
                }
                
                console.log(`  ✓ Cleaned ${collection.name}`);
            } catch (error) {
                console.log(`  ⚠️  Could not clean ${collection.name}:`, error.message);
            }
        }
    }

            // Simulate aggregation in JavaScript
            const courseStats = enrollments.reduce((acc, enrollment) => {
                const courseId = enrollment.course_id;
                if (!acc[courseId]) {
                    acc[courseId] = { total: 0, completed: 0, avgProgress: 0 };
                }
                acc[courseId].total++;
                if (enrollment.is_completed) acc[courseId].completed++;
                acc[courseId].avgProgress += enrollment.progress_percentage || 0;
                return acc;
            }, {});
            
            Object.keys(courseStats).forEach(courseId => {
                courseStats[courseId].avgProgress /= courseStats[courseId].total;
            });
            
            const aggTime = Date.now() - aggStartTime;
            console.log(`  ✓ Aggregation query: ${aggTime}ms (${Object.keys(courseStats).length} courses analyzed)`);
            
        } catch (error) {
            console.log('  ❌ Complex query test failed:', error.message);
        }
    }
